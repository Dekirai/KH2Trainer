native_steps=0
def native_step(data,cursor,acc):
    global native_steps
    def readmem(a,n):
        p=a-0x10000000
        if p<0 or p+n>len(data):raise ValueError('truncated')
        return int.from_bytes(data[p:p+n],'little')
    op=readmem(0x10000000+cursor,2)
    post=i32(cursor+2)
    if op>63:return dict(status='unknown',cursor=post,acc=acc)
    regs={'r8':0x10000000,'rdx':post&0xffffffff,'r10':acc&0xffffffff};pc=targets[op];zero=False;sign=False;steps=0
    def regread(n):
        root,bits=aliases.get(n,(n,64));return regs.get(root,0)&((1<<bits)-1)
    def write(n,value):
        root,bits=aliases.get(n,(n,64));regs[root]=value&((1<<bits)-1)
    def ea(i,o):
        m=o.mem;return (regread(i.reg_name(m.base)) if m.base else 0)+(regread(i.reg_name(m.index))*m.scale if m.index else 0)+m.disp
    def value(i,o):
        if o.type==X86_OP_IMM:return o.imm
        if o.type==X86_OP_REG:return regread(i.reg_name(o.reg))
        return readmem(ea(i,o),o.size)
    while True:
        if pc in [base+0xB7910,base+0xB79D9,base+0xB79EA]:
            native_steps+=steps
            return dict(status={base+0xB7910:'continue',base+0xB79D9:'stop',base+0xB79EA:'unknown'}[pc],cursor=i32(regread('edx')),acc=i32(regread('r10d')))
        i=ins[pc];pc+=i.size;steps+=1
        if steps>1000:raise AssertionError('review model loop bound')
        m=i.mnemonic;o=i.operands
        if m=='nop':continue
        if m=='jmp':pc=value(i,o[0]);continue
        if m in ['jne','jle']:
            if (not zero if m=='jne' else zero or sign):pc=value(i,o[0])
            continue
        if m=='test':
            v=value(i,o[0])&value(i,o[1]);zero=v==0;sign=bool(v&(1<<(o[0].size*8-1)));continue
        assert o[0].type==X86_OP_REG
        name=i.reg_name(o[0].reg)
        if m in ['mov','movzx']:v=value(i,o[1])
        elif m in ['movsx','movsxd']:
            v=value(i,o[1]);bits=o[1].size*8;v=(v&((1<<(bits-1))-1))-(v&(1<<(bits-1)))
        elif m=='lea':v=ea(i,o[1])
        elif m=='add':v=value(i,o[0])+value(i,o[1])
        elif m=='sub':v=value(i,o[0])-value(i,o[1]);zero=(v&((1<<(o[0].size*8))-1))==0
        else:raise AssertionError(('unmodeled',m))
        write(name,v)
