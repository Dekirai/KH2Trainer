#define WIN32_LEAN_AND_MEAN
#define NOMINMAX
#include <windows.h>
#include <stdio.h>
#include "../../src/KH2Trainer.Bridge/MotionObserverSupport.h"
// Focused independent regression for the explicit revised callback policy.
// Typed synthetic originals and Windows SEH only; no game code or installation.
constexpr DWORD Primary=0xe0631101,Inner=0xe0631102,Callback=0xe0631103;
enum class Mode { BeforeNested,AfterNested,UnwindNested,BeforeDirect,AfterDirect,UnwindDirect };
motion_observer::Observer Owners[6];
struct Case { motion_observer::OriginalFn entry;Mode mode;unsigned before[2]{},original[2]{},after[2]{};bool badFlags=false; };
static Case* current=nullptr;
static bool IsUnwind(Mode m){return m==Mode::UnwindNested||m==Mode::UnwindDirect;}
uint64_t __fastcall Before(void*,uintptr_t c){
    ++current->before[c];
    if(c==0&&current->mode==Mode::BeforeNested)current->entry(1,19,1.0f,2.0f);
    if(c==0&&current->mode==Mode::BeforeDirect)RaiseException(Callback,0,0,nullptr);
    return c+99;
}
uint8_t __fastcall Original(uintptr_t c,int32_t,float,float){
    ++current->original[c];
    if(c==1)RaiseException(Inner,0,0,nullptr);
    if(IsUnwind(current->mode))RaiseException(Primary,0,0,nullptr);
    return 0x81;
}
void __fastcall After(void*,uintptr_t c,uint64_t ticket,bool completed,bool normal){
    ++current->after[c];
    const bool beforeFailed=c==0&&(current->mode==Mode::BeforeNested||current->mode==Mode::BeforeDirect);
    if(ticket!=(beforeFailed?0:c+99)||completed==beforeFailed||normal!=(c==0&&!beforeFailed&&!IsUnwind(current->mode)))current->badFlags=true;
    if(c==1)return;
    if(current->mode==Mode::AfterNested||current->mode==Mode::UnwindNested)current->entry(1,19,1.0f,2.0f);
    if(current->mode==Mode::AfterDirect||current->mode==Mode::UnwindDirect)RaiseException(Callback,0,0,nullptr);
}
DWORD Invoke(uint8_t& result){
    __try {result=current->entry(0,-3,3.0f,4.0f);}
    __except(EXCEPTION_EXECUTE_HANDLER){return GetExceptionCode();}
    return 0;
}
template<unsigned I>uint8_t __fastcall Entry(uintptr_t c,int32_t s,float b,float f){return Owners[I].Select(c,s,b,f);}
int main(){
    motion_observer::OriginalFn entries[]={Entry<0>,Entry<1>,Entry<2>,Entry<3>,Entry<4>,Entry<5>};
    unsigned checks=0,failures=0;
    const auto check=[&](bool ok,const char* label){++checks;if(!ok){++failures;printf("FAIL %s\n",label);}};
    for(unsigned i=0;i<6;++i){
        Case c{entries[i],static_cast<Mode>(i)};current=&c;
        check(Owners[i].Bind(Original,Before,After,nullptr),"bind reviewed policy fixture");
        uint8_t result=0xee;const auto caught=Invoke(result);
        const bool before=i==0||i==3,nested=i<=2;
        const DWORD expected=IsUnwind(c.mode)?Primary:nested?Inner:Callback;
        check(caught==expected,"exception identity follows explicit propagation/primary policy");
        check(result==0xee,"abnormal call never fabricates a returned AL value");
        check(c.before[0]==1&&c.original[0]==(before?0u:1u)&&c.after[0]==1,"outer callback/original counts fit failure boundary; After runs once even on Before failure");
        check(c.before[1]==(nested?1u:0u)&&c.original[1]==(nested?1u:0u)&&c.after[1]==(nested?1u:0u),"nested original and finally run once when requested");
        check(!c.badFlags,"all completed invocation tickets and normal flags agree");
        check(Owners[i].Faulted(),"every abnormal invocation latches fault");
        printf("policyCase=%u caught=%08lx expected=%08lx outer=%u/%u/%u inner=%u/%u/%u\n",i,caught,expected,c.before[0],c.original[0],c.after[0],c.before[1],c.original[1],c.after[1]);
    }
    printf("CallbackPolicyReviewTests: %u checks, %u failures. Six synthetic exception-policy cases.\n",checks,failures);
    return failures?1:0;
}
