# Independent mathematical witnesses for the finite fixture geometry. This is

def inside_xz(x,z,verts):
    signs=[]
    for (a,bb),(c,d) in zip(verts,verts[1:]+verts[:1]):signs.append((c-a)*(z-bb)-(d-bb)*(x-a))
    return all(v>=0 for v in signs) or all(v<=0 for v in signs)
