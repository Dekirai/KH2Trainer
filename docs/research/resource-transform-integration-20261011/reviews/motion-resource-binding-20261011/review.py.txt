# Independent control-flow checks. Calls are opaque returning nodes; exceptions,
# callee reentrancy and the actual game scheduler are outside this local graph.
def graph(entry):
 ins=fnins[BASE+entry];g={pc:set() for pc in ins}
 for pc,i in ins.items():
  nxt=pc+i.size
  if i.mnemonic.startswith('ret'):continue
  if i.mnemonic=='jmp':
   if i.operands[0].type==X86_OP_IMM and i.operands[0].imm in ins:g[pc].add(i.operands[0].imm)
  elif i.mnemonic.startswith('j'):
   need(i.operands[0].type==X86_OP_IMM,'known conditional edge')
   if i.operands[0].imm in ins:g[pc].add(i.operands[0].imm)
   if nxt in ins:g[pc].add(nxt)
  elif nxt in ins:g[pc].add(nxt)
 return g
def reachable(g,start,goal,removed=None):
 pending=[start];seen=set()
 while pending:
  pc=pending.pop()
  if pc==removed or pc in seen:continue
  if pc==goal:return True
  seen.add(pc);pending+=list(g.get(pc,()))
